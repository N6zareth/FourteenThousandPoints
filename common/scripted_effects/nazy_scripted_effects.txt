nazy_reset_golden_age = {
	hidden_effect = {
		if = {
			limit = {
				AND = {
					OR = {
						has_country_flag = economic_golden_age
						has_country_flag = military_golden_age
					}
					AND = {
						NOT = { has_country_modifier = economic_golden_age }
						NOT = { has_country_modifier = military_golden_age }
					}	
				}
			}
			clr_country_flag = economic_golden_age
			clr_country_flag = military_golden_age
			}
		else_if = {
			limit = {
				AND = {
					NOT = { has_country_flag = economic_golden_age }
					NOT = { has_country_flag = military_golden_age }
				}
				OR = {
					has_country_modifier = economic_golden_age
					has_country_modifier = military_golden_age
				}
			}
			extend_country_modifier = {
				name = economic_golden_age
				duration = 18250
			}
			extend_country_modifier = {
				name = military_golden_age
				duration = 18250
			}
		}
	}
	custom_tooltip = nazy_reset_golden_age_tt
}
